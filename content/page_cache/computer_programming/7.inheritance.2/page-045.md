Comparable<T> Example: Sorting

 ArrayList<Student> sList = new ArrayList<Student>();

 Student s1 = new Student(3.3);
 Student s2 = new Student(4.2);
 Student s3 = new Student(3.1);

 sList.add(s1);
 sList.add(s2);
 sList.add(s3);

 Collections.sort(sList);
 System.out.println(sList); // [3.1, 3.3, 4.2]
