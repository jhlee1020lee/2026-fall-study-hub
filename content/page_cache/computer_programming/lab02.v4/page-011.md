String[] cars = { "Volvo", "BMW", "Ford", "Mazda" };

for (int i = 0; i < 4; i++) {
  System.out.print(cars[i] + " ");
}
System.out.println();

for (String car : cars) { System.out.print(car + " "); }
System.out.println();
