Increment and Decrement Operators

 int x1 = 1, y1 = 1;
 int prefixResult = x1 + ++y1;
 System.out.println("Prefix result is: " + prefixResult);
 // Prefix result is: 3

 int x2 = 1, y2 = 1;
 int postfixResult = x2 + y2++;
 System.out.println("Postfix result is: " + postfixResult);
 // Postfix result is: 2
